' ============================================================================
'  Worksheet code-behind for the "Company research" sheet (RR4 workbook).
' ----------------------------------------------------------------------------
'  This is NOT a .bas module -- it lives in the sheet's own code module.
'  Paste it there via the VBE (right-click the "Company research" sheet in the
'  Project Explorer > View Code), or inject it with CodeModule.AddFromString
'  against the sheet's CodeName (same mechanism as
'  SEC-Filing-Fetcher/build.ps1's SheetInput_Code.txt).
'
'  Three entry points live here:
'
'  1. Group keyword (upper half). Typing one or more group names into A4
'     (merged A4:B4, dropdown fed by the GroupList name) runs the MARKET
'     SCANNER over those groups via Sanner.RunGroupKeyword. Names must match a
'     group in the Groups sheet's tblGroups exactly (case and spaces ignored);
'     separate several with commas. Status/candidates are written to A5.
'
'  2. Financial deep-dive (lower half). Mirrors SEC-Filing-Fetcher's
'     SheetInput Worksheet_Change: typing a ticker into B108 (or changing the
'     D108 market override) runs CompanyResearchSEC.RunDeepDive.
'
'  3. Group database panel (L1:V33, built by Sanner.RebuildGroupDb).
'     Double-clicking a group name there writes it into A4, which runs
'     trigger 1 above.
'
'  Layout: scan table C1:K33 | group panel L1:V33 | price-chart wall rows
'  35-106 (one line chart per scanned ticker) | deep-dive from row 108.
'
'  Cells:
'    A4   group keyword(s)      (a tblGroups group name, or "name1,name2")
'    A5   group status line     (written by Sanner.RunGroupKeyword)
'    B108 ticker / TW co_id     (e.g. NVDA  or  2395)
'    D108 market override       US | TW | (blank = AUTO-detect)
'    F108 status / error line   (written by CompanyResearchSEC)
' ============================================================================

Private Sub Worksheet_BeforeDoubleClick(ByVal Target As Range, Cancel As Boolean)
    If Sanner.RunGroupFromDb(Target) Then Cancel = True
End Sub

Private Sub Worksheet_Change(ByVal Target As Range)
    On Error GoTo Done

    Dim hitInput As Boolean, hitMarket As Boolean, hitGroup As Boolean
    hitInput = Not Intersect(Target, Me.Range(CompanyResearchSEC.CR_INPUT_CELL)) Is Nothing
    hitMarket = Not Intersect(Target, Me.Range(CompanyResearchSEC.CR_MARKET_CELL)) Is Nothing
    hitGroup = Not Intersect(Target, Me.Range(Sanner.GROUP_INPUT_CELL)) Is Nothing
    If Not (hitInput Or hitMarket Or hitGroup) Then Exit Sub

    If hitGroup Then
        ' The scan writes hundreds of cells on this sheet; keep them from
        ' re-entering this handler one by one.
        Application.EnableEvents = False
        Call Sanner.RunGroupKeyword(CStr(Me.Range(Sanner.GROUP_INPUT_CELL).Value))
        Application.EnableEvents = True
        Exit Sub
    End If

    Application.EnableEvents = False

    Dim tk As String, mkt As String
    tk = Trim$(CStr(Me.Range(CompanyResearchSEC.CR_INPUT_CELL).Value))
    mkt = Trim$(CStr(Me.Range(CompanyResearchSEC.CR_MARKET_CELL).Value))

    Application.EnableEvents = True

    Call CompanyResearchSEC.RunDeepDive(tk, mkt)
    Exit Sub

Done:
    Application.EnableEvents = True
End Sub
