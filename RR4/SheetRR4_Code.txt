' ================================================================
'  Code-behind of the RR4 worksheet (sheet code name differs per
'  workbook - look the sheet up by its tab name "RR4"). Kept here as
'  plain text because document modules are not exported as .bas.
'
'  Worksheet_Change: the page's own input cells -
'    D5  ARRANGE <GO>   -> PortfolioDashboard_v3.ApplyArrange
'    J4  TICKER <GO>    -> TickerInsight.RefreshTickerInsight
'    K20 PRICE TARGET   -> TickerInsight.RefreshTickerProjection
'    (WATCHLIST entry row retired 2026-09-25 - names are typed on the Watch
'     page, modWatch; since 2026-09-27 the RR4 WATCHLIST is a display-only
'     list from row 58 with NO events at all)
'    R/S UPSIDE / DOWNSIDE of a position row -> RecalcTargetRow (2026-09-25;
'                        the workbook is in manual calculation, so the P.TARGET /
'                        SWING RISK formulas are refreshed by hand here)
'    TO-DO entry row (G59, H59:K59 merged, L59) -> PortfolioDashboard_v3.TodoCommitEntry
'                        (2026-09-21, moved 2026-09-27 from B37:D37; commits on
'                        Enter in the due cell, or in the task cell when the due
'                        cell already holds a date. The task cell is MERGED, so
'                        Target is the whole H:K area - CountLarge > 1 is only
'                        tolerated for a merged Target)
'  Worksheet_BeforeDoubleClick: G60:M61 saved TO-DO rows -> TodoDeleteRow (done)
'                               (the WATCHLIST jump to the Watch page was
'                               removed 2026-09-27)
'  (addresses come from the RR4_* / TI_* constants). The nav command
'  cell B1 is handled workbook-wide in ThisWorkbook.Workbook_SheetChange.
'  The CmdBtn0-7 click handlers were removed with the ActiveX buttons
'  (2026-09-12) - their jobs are nav commands now (see modNav).
'  Worksheet_Activate/Deactivate (2026-09-23): staying on this sheet for
'  60s starts a 15s live-price auto-refresh loop (modLiveRefresh); leaving
'  cancels it. See modLiveRefresh.bas / PortfolioDashboard_v3.RefreshLivePricesLite.
' ================================================================
Private Sub Worksheet_Change(ByVal Target As Range)
    If Target.CountLarge > 1 Then
        If Not Target.MergeCells Then Exit Sub
        If Intersect(Target, Me.Range(Me.Cells(RR4_TD_ENTRY, RR4_TD_C_TASK), _
                                      Me.Cells(RR4_TD_ENTRY, RR4_TD_C_TASK2))) Is Nothing Then Exit Sub
    End If
    ' UPSIDE / DOWNSIDE typed in the position log -> refresh that row's P.TARGET / SWING RISK
    If Target.Row >= RR4_POS_FIRST And Target.Row < RR4_POS_FIRST + 100 And _
       (Target.Column = RR4_UP_COL Or Target.Column = RR4_DN_COL) Then
        Call RecalcTargetRow(Me, Target.Row)
        Exit Sub
    End If
    Dim hitArr As Boolean, hitTicker As Boolean, hitTarget As Boolean, hitTD As Boolean
    hitArr = Not Intersect(Target, Me.Range(RR4_ARR_CELL)) Is Nothing
    hitTicker = Not Intersect(Target, Me.Range(TI_TICKER_CELL)) Is Nothing
    hitTarget = Not Intersect(Target, Me.Range(TI_TARGET_CELL)) Is Nothing
    hitTD = Not Intersect(Target, Me.Range(Me.Cells(RR4_TD_ENTRY, RR4_TD_C_TICK), _
                                            Me.Cells(RR4_TD_ENTRY, RR4_TD_C_DUE))) Is Nothing
    If Not (hitArr Or hitTicker Or hitTarget Or hitTD) Then Exit Sub

    Application.EnableEvents = False
    On Error GoTo Fin
    If hitArr Then
        Call ApplyArrange(CStr(Target.Value))
    ElseIf hitTicker Then
        Call RefreshTickerInsight(CStr(Target.Value))
    ElseIf hitTarget Then
        Call RefreshTickerProjection
    Else
        Call TodoCommitEntry(Me, Target.Column)
    End If

Fin:
    Application.EnableEvents = True
    Application.ScreenUpdating = True
    If Err.Number <> 0 Then
        MsgBox "RR4 input error " & Err.Number & ": " & Err.Description, vbCritical
    End If
End Sub

Private Sub Worksheet_BeforeDoubleClick(ByVal Target As Range, Cancel As Boolean)
    If Target.CountLarge > 1 Then
        If Not Target.MergeCells Then Exit Sub          ' merged task cell H:K reports its whole area
    End If
    If Intersect(Target, Me.Range(Me.Cells(RR4_TD_FIRST, RR4_TD_C_TICK), _
                                  Me.Cells(RR4_TD_LAST, RR4_TD_C_DTE))) Is Nothing Then Exit Sub
    Cancel = True
    Application.EnableEvents = False
    On Error GoTo Fin
    Call TodoDeleteRow(Me, Target.Row)
Fin:
    Application.EnableEvents = True
    If Err.Number <> 0 Then MsgBox "To-do delete error " & Err.Number & ": " & Err.Description, vbCritical
End Sub

Private Sub Worksheet_Activate()
    Call modLiveRefresh.StartDwellTimer
End Sub

Private Sub Worksheet_Deactivate()
    Call modLiveRefresh.StopAutoRefresh
End Sub
